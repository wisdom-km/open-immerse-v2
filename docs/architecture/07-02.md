### 7.2 Python 侧引擎接口（worker 内部）

```python
# workers/parser/oi_worker/engines/base.py —— 拟议
from typing import Protocol, Iterable
from oi_worker.protocol import (PageInput, PageCandidate, EngineCapabilities,
                                LayoutRegion, LayoutResult, ReadingOrderResult,
                                FormulaCandidate, TextSpan, CancelToken,
                                EvalItem, EvalOutcome)

class DocumentEngine(Protocol):
    """识别阶段：把一页（或已有版面区域）变成候选页。引擎 A 的识别阶段与引擎 B 各自实现它。版面不在这次调用里。"""
    def capabilities(self) -> EngineCapabilities: ...
    def parse_page(self, page: PageInput, cancel: CancelToken) -> PageCandidate: ...

# 版面是独立作业，与引擎 A 的识别阶段分开（Wisdom 2026-10-30 23:55）。
# 本地与远程实现同一个 Protocol，走同一套 HTTP + SSE（§10.2 的 layout 作业）。
class LayoutEngine(Protocol):
    """输入页面图，输出带坐标和阅读顺序的区块。与 QualityEvaluatorBackend 同级，不是 parse_page 的内部步骤。"""
    def describe(self) -> LayoutDescriptor: ...   # 实现 id、部署（local / remote）、模型版本
    def detect(self, page: PageInput, cancel: CancelToken) -> LayoutResult: ...
    # PageInput: 页面图 ArtifactRef + CoordinateFrame（不传 PDF 路径）
    # LayoutResult: regions（id、label、bbox，0–1 归一化、左上原点）+ model_order（模型没给出顺序时为 None，§7.10）

# 引擎 A 的识别阶段由以下可替换部件组合，每个部件一个文件、一个实现类。
# 它们接收版面作业已经给出的 LayoutResult，不再自己做版面。
# 没有版面作业时不调用引擎 A 的 parse_page，也不启动引擎 B：引擎 B 的 SDK 内部会跑 PP-DocLayoutV3，跑它等于悄悄把版面加回来（D-E，Wisdom 2026-10-31 11:14）。
class LayoutDetector(Protocol):         # 只是 LayoutEngine 的 worker 侧名字，实现放在 layout 作业里，不放进 parse_page
    def detect(self, page: PageInput) -> LayoutResult: ...
class TextExtractor(Protocol):          # 文字层对齐优先，OCR 兜底
    def extract(self, page: PageInput, regions: list[LayoutRegion]) -> list[TextSpan]: ...
class FormulaRecognizer(Protocol):
    def recognize(self, page: PageInput, regions: list[LayoutRegion]) -> list[FormulaCandidate]: ...
class TableRecognizer(Protocol):
    def recognize(self, page: PageInput, regions: list[LayoutRegion]) -> list[LayoutRegion]: ...
class ReadingOrderer(Protocol):         # 校验模型顺序并兜底（§7.10）；规则集版本进入产物
    ruleset_version: str                # 如 "oi-rules/1"
    def order(self, regions: list[LayoutRegion], model_order: list[str] | None) -> ReadingOrderResult: ...

# 质量评估后端（§7.4、Q9）：本机推理与远程 HTTP 两种实现，对 api/ 暴露同一个接口
class QualityEvaluatorBackend(Protocol):
    max_batch: int                      # Jev 为 64
    def describe(self) -> EvaluatorDescriptor: ...   # 部署方式、模型版本、渲染环境、template_md5 → evaluatorVersion
    def evaluate(self, items: list[EvalItem]) -> list[EvalOutcome]: ...
    # EvalItem: 原区域图 + 候选文本 + 类别（text / formula / table）+ 可选渲染图
    # EvalOutcome: 上游原始 verdict 与原始分，原样返回；规范化在 TS 的 quality/domain 完成
```
